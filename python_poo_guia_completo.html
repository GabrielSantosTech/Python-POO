class — Orientação a Objetos básica

O que é uma classe? Uma classe é um molde. Um objeto é a coisa criada a partir desse molde. __init__ é o construtor — roda automaticamente quando você cria o objeto. self representa o próprio objeto dentro da classe.
📦 Primeira classe — do zero
python
class Pessoa:
    # __init__ é chamado ao criar o objeto
    def __init__(self, nome: str, idade: int):
        self.nome  = nome   # atributo de instância
        self.idade = idade

    # Método de instância
    def apresentar(self):
        print(f"Oi! Sou {self.nome}, {self.idade} anos.")

    def aniversario(self):
        self.idade += 1
        print(f"Parabéns {self.nome}! Agora tem {self.idade} anos.")

    def eh_maior(self) -> bool:
        return self.idade >= 18


# Criando objetos (instâncias)
p1 = Pessoa("Ana", 25)
p2 = Pessoa("João", 15)

p1.apresentar()          # Oi! Sou Ana, 25 anos.
p1.aniversario()         # Parabéns Ana! Agora tem 26 anos.
print(p2.eh_maior())     # False

# Acessar e modificar atributos diretamente
print(p1.nome)            # Ana
p1.nome = "Ana Paula"
print(p1.nome)            # Ana Paula
🏷️ Atributos de classe vs atributos de instância
⚠️ Atributos de classe são compartilhados entre todos os objetos. Atributos de instância (self.x) pertencem a cada objeto individualmente.
python
class Contador:
    total = 0  # atributo de CLASSE — compartilhado por todos

    def __init__(self, nome):
        self.nome = nome          # atributo de INSTÂNCIA
        Contador.total += 1

    @classmethod
    def quantos(cls):
        print(f"Total criados: {cls.total}")


c1 = Contador("primeiro")
c2 = Contador("segundo")
c3 = Contador("terceiro")
Contador.quantos()    # Total criados: 3
print(Contador.total)  # 3 — mesmo valor para todos

class Produto:
    def __init__(self, nome: str, preco: float, estoque: int = 0):
        self.nome    = nome
        self.preco   = preco
        self.estoque = estoque

    def adicionar(self, qtd: int):
        self.estoque += qtd
        print(f"+{qtd} unidades de {self.nome}. Estoque: {self.estoque}")

    def vender(self, qtd: int):
        if qtd > self.estoque:
            print(f"Estoque insuficiente! Temos {self.estoque} unidades.")
        else:
            self.estoque -= qtd
            total = qtd * self.preco
            print(f"Venda: {qtd}x {self.nome} = R${total:.2f}")

    def info(self):
        print(f"[{self.nome}] R${self.preco:.2f} | Estoque: {self.estoque}")


cafe  = Produto("Café 500g", 18.90)
leite = Produto("Leite 1L", 6.50, estoque=20)

cafe.adicionar(10)
cafe.vender(3)
cafe.info()
leite.vender(25)   # vai dar erro de estoque

Herança

O que é herança? A classe filha herda todos os atributos e métodos da classe pai. Use super() para acessar o pai de dentro da filha. Você pode sobrescrever (override) qualquer método.
🧬 Herança simples
python
class Animal:
    def __init__(self, nome: str, idade: int):
        self.nome  = nome
        self.idade = idade

    def apresentar(self):
        print(f"Animal: {self.nome}, {self.idade} anos")

    def falar(self):
        print("... (sem som)")


# Cachorro HERDA de Animal
class Cachorro(Animal):
    def __init__(self, nome, idade, raca):
        super().__init__(nome, idade)  # chama o __init__ do pai
        self.raca = raca               # atributo próprio

    def falar(self):             # OVERRIDE — sobrescreve o pai
        print(f"{self.nome}: Au au!")

    def buscar(self):            # método só do Cachorro
        print(f"{self.nome} foi buscar a bolinha!")


class Gato(Animal):
    def falar(self):
        print(f"{self.nome}: Miau!")

    def ronronar(self):
        print(f"{self.nome} está ronronando... 😻")


rex  = Cachorro("Rex", 3, "Labrador")
mimi = Gato("Mimi", 5)

rex.apresentar()   # herdado do Animal
rex.falar()        # Rex: Au au! (override)
rex.buscar()       # método próprio
mimi.falar()       # Mimi: Miau!

# Verificar herança
print(isinstance(rex, Animal))    # True
print(isinstance(rex, Cachorro)) # True
print(issubclass(Gato, Animal))  # True

Encapsulamento

Convenções de acesso em Python nome = público (qualquer um acessa)  |  _nome = protegido (convenção: não acesse de fora)  |  __nome = privado (Python aplica name mangling)
🔒 _ e __ na prática
python
class ContaBancaria:
    def __init__(self, titular, saldo_inicial=0):
        self.titular   = titular         # público
        self._agencia  = "0001"          # protegido (convenção)
        self.__saldo   = saldo_inicial   # privado (name mangling)

    def depositar(self, valor):
        if valor > 0:
            self.__saldo += valor
            print(f"Depósito de R${valor:.2f}. Novo saldo: R${self.__saldo:.2f}")
        else:
            print("Valor inválido!")

    def sacar(self, valor):
        if valor > self.__saldo:
            print("Saldo insuficiente!")
        else:
            self.__saldo -= valor
            print(f"Saque de R${valor:.2f}. Saldo: R${self.__saldo:.2f}")

    def ver_saldo(self):
        print(f"Saldo de {self.titular}: R${self.__saldo:.2f}")


conta = ContaBancaria("Ana", 1000)
conta.depositar(500)
conta.sacar(200)
conta.ver_saldo()

🏷️ @property — getters e setters elegantes
💡 @property permite acessar um método como se fosse um atributo, e @x.setter controla a atribuição com validação.
python
class Pessoa:
    def __init__(self, nome, idade):
        self.nome  = nome
        self.__idade = idade   # privado

    @property
    def idade(self):          # getter — acesso como atributo
        return self.__idade

    @idade.setter
    def idade(self, valor):   # setter — com validação
        if not isinstance(valor, int) or valor < 0:
            raise ValueError("Idade deve ser inteiro positivo!")
        self.__idade = valor

    @property
    def eh_maior(self) -> bool:
        return self.__idade >= 18


p = Pessoa("Carlos", 20)
print(p.idade)       # 20 — parece atributo, mas é método!
print(p.eh_maior)    # True
p.idade = 25        # usa o setter com validação
print(p.idade)       # 25

try:
    p.idade = -5    # vai lançar ValueError
except ValueError as e:
    print(f"Erro: {e}")

📋 Lista de objetos — o padrão mais usado

class Turma:
    """Gerencia uma lista de alunos."""
    def __init__(self, nome):
        self.nome   = nome
        self.alunos = []    # lista de objetos Aluno

    def adicionar(self, aluno):
        self.alunos.append(aluno)
        print(f"✅ {aluno.nome} adicionado(a) à turma {self.nome}")

    def listar(self):
        print(f"\n📚 Turma: {self.nome} ({len(self.alunos)} alunos)")
        for i, aluno in enumerate(self.alunos, 1):
            aluno.exibir(i)

    def buscar(self, nome) -> list:
        return [a for a in self.alunos if nome.lower() in a.nome.lower()]

    def media_geral(self):
        notas = [a.nota for a in self.alunos if a.nota is not None]
        return sum(notas) / len(notas) if notas else 0

    def aprovados(self):
        return [a for a in self.alunos if (a.nota or 0) >= 6]


class Aluno:
    def __init__(self, nome, nota=None):
        self.nome = nome
        self.nota = nota

    def exibir(self, pos=""):
        nota_str = f"{self.nota:.1f}" if self.nota is not None else "sem nota"
        status   = "✅" if (self.nota or 0) >= 6 else "❌"
        print(f"  {pos}. {self.nome} — {nota_str} {status}")

def menu_turma():
    turma = Turma(input("Nome da turma: "))

    while True:
        print("\n📚 GESTÃO DE TURMA")
        print("1. Adicionar aluno")
        print("2. Listar alunos")
        print("3. Buscar aluno")
        print("4. Ver aprovados")
        print("0. Sair")
        op = input("Opção: ")

        if op == "1":
            nome = input("Nome do aluno: ")
            try:
                nota = float(input("Nota (Enter para pular): ") or "0")
            except:
                nota = None
            turma.adicionar(Aluno(nome, nota))

        elif op == "2":
            turma.listar()
            print(f"  Média geral: {turma.media_geral():.1f}")

        elif op == "3":
            nome = input("Buscar por nome: ")
            found = turma.buscar(nome)
            if found:
                for a in found: a.exibir()
            else:
                print("Nenhum aluno encontrado.")

        elif op == "4":
            ap = turma.aprovados()
            print(f"Aprovados: {len(ap)}/{len(turma.alunos)}")
            for a in ap: a.exibir()

        elif op == "0":
            print("Até logo!")
            break

menu_turma()

Métodos Especiais (Dunder Methods)

ℹ️ Métodos com duplo underscore (__metodo__) são chamados de dunder methods ou métodos mágicos. Python os chama automaticamente em situações específicas.
✨ __str__, __repr__, __len__, __eq__
python
class Livro:
    def __init__(self, titulo, autor, paginas):
        self.titulo  = titulo
        self.autor   = autor
        self.paginas = paginas

    def __str__(self):           # print(livro) e str(livro)
        return f'"{self.titulo}" por {self.autor}'

    def __repr__(self):          # repr(livro) — dev-friendly
        return f'Livro("{self.titulo}", "{self.autor}", {self.paginas})'

    def __len__(self):           # len(livro)
        return self.paginas

    def __eq__(self, outro):     # livro1 == livro2
        return self.titulo == outro.titulo and self.autor == outro.autor

    def __lt__(self, outro):     # livro1 < livro2 (para sorted)
        return self.paginas < outro.paginas

    def __add__(self, outro):    # livro1 + livro2 = total de páginas
        return self.paginas + outro.paginas

    def __contains__(self, palavra): # "python" in livro
        return palavra.lower() in self.titulo.lower()


l1 = Livro("Python Crash Course", "Eric Matthes", 544)
l2 = Livro("Fluent Python", "Luciano Ramalho", 792)

print(l1)                   # usa __str__
print(repr(l1))             # usa __repr__
print(len(l1))              # 544 — usa __len__
print(l1 == l2)             # False — usa __eq__
print(l1 + l2)              # 1336 — usa __add__
print("Python" in l1)      # True — usa __contains__

livros = [l1, l2]
print(sorted(livros))       # ordena por páginas — usa __lt__
🔁 __iter__ e __next__ — tornar iterável
python
class Contagem:
    """Conta de inicio até fim — pode usar em for!"""
    def __init__(self, inicio, fim):
        self.inicio = inicio
        self.fim    = fim
        self.atual  = inicio

    def __iter__(self):     # for x in contagem:
        self.atual = self.inicio
        return self

    def __next__(self):     # próximo item do for
        if self.atual > self.fim:
            raise StopIteration
        valor = self.atual
        self.atual += 1
        return valor


for n in Contagem(1, 5):
    print(n, end=" ")   # 1 2 3 4 5

# Também funciona com list(), sum(), etc.
print(list(Contagem(1, 10)))
print(sum(Contagem(1, 100)))  # 5050

🎭 Polimorfismo — duck typing
Duck typing "Se anda como pato e grasna como pato, é um pato." — em Python, não importa o tipo, importa se o objeto tem o método.

class Cachorro:

# Polimorfismo — não importa o tipo, só precisa ter .falar()
animais = [Cachorro(), Gato(), Papagaio(), Cachorro()]
for animal in animais:

🏷️ @classmethod e @staticmethod
python
class Data:
    def __init__(self, dia, mes, ano):
        self.dia = dia
        self.mes = mes
        self.ano = ano

    def __str__(self):
        return f"{self.dia:02d}/{self.mes:02d}/{self.ano}"

    @classmethod
    def de_string(cls, texto: str):
        # construtor alternativo — cria Data de "dd/mm/aaaa"
        d, m, a = map(int, texto.split("/"))
        return cls(d, m, a)

    @classmethod
    def hoje(cls):
        from datetime import date
        d = date.today()
        return cls(d.day, d.month, d.year)

    @staticmethod
    def eh_bissexto(ano: int) -> bool:
        # não usa self nem cls — função utilitária relacionada
        return ano % 4 == 0 and (ano % 100 != 0 or ano % 400 == 0)


d1 = Data(25, 12, 2025)
d2 = Data.de_string("01/01/2026")  # @classmethod
d3 = Data.hoje()                   # @classmethod
print(d1, d2, d3)
print(Data.eh_bissexto(2024))      # True — @staticmethod
📐 ABC — Classes Abstratas
💡 Classes abstratas definem uma interface: obrigam as filhas a implementar certos métodos. Não dá pra instanciar a classe abstrata diretamente.
python
from abc import ABC, abstractmethod

class Forma(ABC):       # classe abstrata
    def __init__(self, cor="branco"):
        self.cor = cor

    @abstractmethod
    def area(self) -> float:
        pass   # OBRIGATÓRIO implementar nas filhas

    @abstractmethod
    def perimetro(self) -> float:
        pass

    def info(self):       # método concreto — já implementado
        print(f"Área: {self.area():.2f} | Perímetro: {self.perimetro():.2f}")


class Circulo(Forma):
    def __init__(self, raio, cor="branco"):
        super().__init__(cor)
        self.raio = raio

    def area(self):
        import math
        return math.pi * self.raio ** 2

    def perimetro(self):
        import math
        return 2 * math.pi * self.raio


class Retangulo(Forma):
    def __init__(self, l, a): self.l, self.a = l, a
    def area(self):       return self.l * self.a
    def perimetro(self): return 2 * (self.l + self.a)


formas = [Circulo(5), Retangulo(4, 6), Circulo(3)]
for f in formas:
    f.info()   # polimorfismo + ABC

# Forma() → TypeError: Can't instantiate abstract class
📦 Dataclasses — classes de dados sem boilerplate

from dataclasses import dataclass, field

@dataclass
class Produto:
    nome:    str
    preco:   float
    estoque: int  = 0
    tags:    list = field(default_factory=list)

    def com_desconto(self, pct: float) -> float:
        return self.preco * ((100 - pct) / 100)


# @dataclass gera __init__, __repr__, __eq__ automaticamente!

Projeto completo — Sistema de Biblioteca

🎯 Um sistema completo com herança, encapsulamento, listas de objetos, input e métodos especiais.
🏗️ Código completo
python
from datetime import date

# ── CLASSES BASE ──────────────────────────────────────────
class Pessoa:
    def __init__(self, nome: str, email: str):
        self.nome  = nome
        self.email = email

    def __repr__(self):
        return f"{self.__class__.__name__}({self.nome!r})"


class Membro(Pessoa):
    _prox_id = 1

    def __init__(self, nome, email):
        super().__init__(nome, email)
        self.id       = Membro._prox_id
        Membro._prox_id += 1
        self.emprestimos = []

    def pegar_livro(self, livro):
        self.emprestimos.append(livro)
        livro.disponivel = False
        print(f"📖 {self.nome} pegou '{livro.titulo}'")

    def devolver(self, livro):
        if livro in self.emprestimos:
            self.emprestimos.remove(livro)
            livro.disponivel = True
            print(f"✅ {self.nome} devolveu '{livro.titulo}'")
        else:
            print("Livro não está com este membro!")

    def listar_emprestimos(self):
        if not self.emprestimos:
            print(f"{self.nome} não tem livros.")
        else:
            print(f"Livros com {self.nome}:")
            for l in self.emprestimos:
                print(f"  • {l.titulo}")


class Livro:
    def __init__(self, titulo: str, autor: str, ano: int):
        self.titulo     = titulo
        self.autor      = autor
        self.ano        = ano
        self.disponivel = True

    def __str__(self):
        status = "✅ disponível" if self.disponivel else "❌ emprestado"
        return f'"{self.titulo}" — {self.autor} ({self.ano}) [{status}]'

class Biblioteca:
    def __init__(self, nome: str):
        self.nome    = nome
        self.livros  = []
        self.membros = []

    def __len__(self):
        return len(self.livros)

    def cadastrar_livro(self):
        titulo = input("Título: ")
        autor  = input("Autor: ")
        try:
            ano = int(input("Ano: "))
        except ValueError:
            ano = date.today().year
        livro = Livro(titulo, autor, ano)
        self.livros.append(livro)
        print(f"📚 Livro '{titulo}' cadastrado!")

    def cadastrar_membro(self):
        nome  = input("Nome do membro: ")
        email = input("Email: ")
        m = Membro(nome, email)
        self.membros.append(m)
        print(f"👤 Membro #{m.id} '{nome}' cadastrado!")

    def listar_livros(self):
        print(f"\n📚 Acervo — {len(self)} livros:")
        for i, l in enumerate(self.livros, 1):
            print(f"  {i}. {l}")

    def listar_membros(self):
        print(f"\n👥 Membros — {len(self.membros)}:")
        for m in self.membros:
            print(f"  #{m.id} {m.nome} ({len(m.emprestimos)} livro(s))")

    def emprestar(self):
        disp = [l for l in self.livros if l.disponivel]
        if not disp: print("Nenhum livro disponível."); return
        print("Livros disponíveis:")
        for i, l in enumerate(disp, 1): print(f"  {i}. {l.titulo}")
        try:
            li = int(input("Número do livro: ")) - 1
            livro = disp[li]
            self.listar_membros()
            mi = int(input("ID do membro: "))
            membro = next((m for m in self.membros if m.id == mi), None)
            if membro: membro.pegar_livro(livro)
            else: print("Membro não encontrado!")
        except (ValueError, IndexError):
            print("Seleção inválida!")

    def menu(self):
        while True:
            print(f"\n🏛️  {self.nome}")
            print("1. Cadastrar livro")
            print("2. Cadastrar membro")
            print("3. Listar livros")
            print("4. Listar membros")
            print("5. Emprestar livro")
            print("0. Sair")
            op = input("Opção: ")

            acoes = {
                "1": self.cadastrar_livro,
                "2": self.cadastrar_membro,
                "3": self.listar_livros,
                "4": self.listar_membros,
                "5": self.emprestar,
            }
            if op in acoes:
                acoes[op]()
            elif op == "0":
                print("Até logo!"); break


# ── RODAR O SISTEMA ───────────────────────────────────────
if __name__ == "__main__":
    nome = input("Nome da biblioteca: ")
    bib  = Biblioteca(nome)
    bib.menu()
